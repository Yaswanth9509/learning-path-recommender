# Runtime dependencies. Upper bounds are deliberate: a major release of any of
# these has broken applications before, and this project is meant to still run
# unchanged a year from now.
fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.7,<3.0

# Postgres. Only used when DATABASE_URL is set; with it unset the app runs on
# stdlib sqlite3 exactly as before and this import never happens. It is a hard
# requirement for deployment, though: a free instance has no persistent disk,
# so a SQLite file is wiped whenever the instance sleeps and every registered
# account goes with it.
psycopg[binary]>=3.1,<4.0
